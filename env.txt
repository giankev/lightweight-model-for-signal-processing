# 1) Remove old environment (if not done)
conda env remove -n radio

# 2) Create new clean environment
conda create -n radio python=3.12 -y
conda activate radio

# 3) Install Jupyter and other environment basics first
conda install jupyterlab ipykernel ipywidgets -y

# 4) Install PyTorch PyPI wheels compiled for CUDA 13.0
pip install --upgrade pip
pip install torch torchvision torchaudio --extra-index-url https://download.pytorch.org/whl/cu130

# 5) Install your other requirements
pip install numpy>=1.21.0 pandas>=1.3.0 matplotlib>=3.4.0 scikit-learn>=0.24.0 tqdm>=4.60.0

# 6) Register kernel
python -m ipykernel install --user --name radio --display-name "radio (py)"
